    配置免费栈

    # 1. 添加免费提供商
omniroute providers add kiro --api-key "你的Kiro密钥"
omniroute providers add qoder         # 无限免费，无需密钥
omniroute providers add pollinations  # 无需密钥
omniroute providers add longcat --api-key "你的LongCat密钥"

# 2. 创建 4 层故障转移组合
omniroute combo create free-stack \
  --model kr/claude-sonnet-4.5 \
  --fallback qd/kimi-k2-thinking \
  --fallback pol/gp-5 \
  --fallback lc/LongCat-2.0 \
  --compression stacked

    CI/CD 集成

    在 GitHub Actions 中使用 OmniRoute：

    # .github/workflows/ai-review.yml
jobs:
  review:
    runs-on: ubuntu-latest
    services:
      omniroute:
        image: diegosouzapw/omniroute:latest
        ports:
          - 20128:20128
    steps:
      - uses: anthropics/claude-code-action@v1
        with:
          apiEndpoint: http://localhost:20128/v1
